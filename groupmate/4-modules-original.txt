// 1. Account Management
async function loadAccounts() {
    const res = await fetch("../api/pos_system_api.php?action=get_users");
    const result = await res.json();
    const tbody = document.getElementById("accountsTableBody");
    if (!tbody) return;

    if (result.status === "success") {
        tbody.innerHTML = result.data.map(u => `
            <tr>
                <td><code>USR-${u.id}</code></td>
                <td class="fw-bold">${u.username}</td>
                <td>${u.email || 'N/A'}</td>
                <td><span class="badge bg-secondary">${u.role}</span></td>
                <td><span class="badge ${u.is_active == 1 ? 'bg-success' : 'bg-danger'}">${u.is_active == 1 ? 'Active' : 'Inactive'}</span></td>
                <td>
                    <button class="btn btn-sm btn-outline-danger" onclick="toggleUser(${u.id})">
                        ${u.is_active == 1 ? 'Deactivate' : 'Activate'}
                    </button>
                </td>
            </tr>
        `).join("");
    }
}

async function toggleUser(userId) {
    await fetch("../api/pos_system_api.php?action=toggle_user", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ user_id: userId })
    });
    loadAccounts();
}

async function handleAddAccount(e) {
    e.preventDefault();
    const payload = {
        username: document.getElementById("accUsername").value,
        full_name: document.getElementById("accFullName").value,
        email: document.getElementById("accEmail").value,
        role: document.getElementById("accRole").value
    };

    await fetch("../api/pos_system_api.php?action=add_user", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
    });
    loadAccounts();
}

// 2. Inventory Management
async function loadInventory() {
    const res = await fetch("../api/pos_system_api.php?action=get_products");
    const result = await res.json();
    const tbody = document.getElementById("inventoryTableBody");
    if (!tbody) return;

    if (result.status === "success") {
        let lowStockCount = 0;
        tbody.innerHTML = result.data.map(p => {
            const isLow = p.stock_qty <= p.reorder_level;
            if (isLow) lowStockCount++;

            return `
                <tr>
                    <td><code>${p.barcode}</code></td>
                    <td class="fw-bold">${p.name}</td>
                    <td>${p.supplier_name || 'N/A'}</td>
                    <td>$${parseFloat(p.price).toFixed(2)}</td>
                    <td>
                        <span class="badge ${isLow ? 'bg-warning text-dark' : 'bg-success'}">
                            ${p.stock_qty} units
                        </span>
                    </td>
                    <td>
                        <button class="btn btn-sm btn-outline-primary me-1" onclick="restockProduct(${p.id}, 10)">+10 Stock</button>
                        <button class="btn btn-sm btn-success" onclick="addCartItem(${p.id}, '${p.name}', ${p.price})">+ POS</button>
                    </td>
                </tr>
            `;
        }).join("");

        document.getElementById("statLowStock").innerText = lowStockCount;
        document.getElementById("statProducts").innerText = result.data.length;
    }
}

async function handleAddProduct(e) {
    e.preventDefault();
    const payload = {
        barcode: document.getElementById("mSku").value,
        name: document.getElementById("mName").value,
        price: parseFloat(document.getElementById("mPrice").value),
        stock_qty: parseInt(document.getElementById("mStock").value),
        supplier_id: document.getElementById("mSupplier").value

    await fetch("../api/pos_system_api.php?action=add_product", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
    });
    loadInventory();
}

// 4. POS / Sales Management
function addCartItem(productId, name, price) {
    const existing = posCart.find(item => item.product_id === productId);
    if (existing) {
        existing.quantity += 1;
    } else {
        posCart.push({ product_id: productId, name, price, quantity: 1 });
    }
    renderPOSCart();
}

function renderPOSCart() {
    const tbody = document.getElementById("posCartBody");
    if (!tbody) return;

    if (posCart.length === 0) {
        tbody.innerHTML = `<tr><td colspan="4" class="text-center text-muted py-4">Cart is empty</td></tr>`;
        document.getElementById("posSubtotal").innerText = "$0.00";
        document.getElementById("posTax").innerText = "$0.00";
        document.getElementById("posTotal").innerText = "$0.00";
        return;
    }

    let subtotal = 0;
    tbody.innerHTML = posCart.map(item => {
        const lineTotal = item.price * item.quantity;
        subtotal += lineTotal;
        return `
            <tr>
                <td>${item.name}</td>
                <td class="text-center">${item.quantity}</td>
                <td class="text-end">$${item.price.toFixed(2)}</td>
                <td class="text-end fw-bold">$${lineTotal.toFixed(2)}</td>
            </tr>
        `;
    }).join("");

    const tax = subtotal * 0.08;
    const total = subtotal + tax;

    document.getElementById("posSubtotal").innerText = `$${subtotal.toFixed(2)}`;
    document.getElementById("posTax").innerText = `$${tax.toFixed(2)}`;
    document.getElementById("posTotal").innerText = `$${total.toFixed(2)}`;
}

async function handlePOSCheckout() {
    if (posCart.length === 0) {
        alert("Please add items to cart before checkout!");
        return;
    }

    const res = await fetch("../api/pos_system_api.php?action=checkout", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ cart: posCart })
    });
    const result = await res.json();

    if (result.status === "success") {
        alert("Transaction completed successfully!");
        posCart = [];
        renderPOSCart();
        loadInventory();
    } else {
        alert("Error: " + result.message);
    }
}

<?php
require_once 'config.php';
session_start();

$data = json_decode(file_get_contents('php://input'), true);

$username = trim($data["username"] ?? '');
$password = $data["password"] ?? ''; 

if (empty($username) || empty($password)) {
    echo json_encode(['success' => false, 'message' => 'Username and password are required.']);
    exit;
}

$stmt = $pdo->prepare("SELECT * FROM users WHERE username = :username AND is_active = 1");
$stmt->execute([':username' => $username]);
$user = $stmt->fetch();

if ($user && password_verify($password, $user['password_hash'])) {
    $_SESSION['user_id']   = $user['id'];
    $_SESSION['username']  = $user['username'];
    $_SESSION['full_name'] = $user['full_name'];
    $_SESSION['role']      = $user['role'];

    $redirectUrl = match($user['role']) {
        'admin'   => 'pages/admin-dashboard.php',
        'manager' => 'pages/manager-dashboard.php',
        'cashier' => 'pages/cashier-dashboard.php',
        default   => 'index.html'
    };

    echo json_encode([
        'success'      => true,
        'role'         => $user['role'],
        'full_name'    => $user['full_name'],
        'redirect_url' => $redirectUrl
    ]);
} else {
    echo json_encode(['success' => false, 'message' => 'Incorrect credentials or account deactivated.']);
}
?>

document.getElementById('loginForm').addEventListener('submit', async function (e) {
  e.preventDefault();

  const usernameInput = document.getElementById('username');
  const passwordInput = document.getElementById('password');
  const roleInput = document.getElementById('role');

  const username = usernameInput.value.trim();
  const password = passwordInput.value;
  const role = roleInput.value;
  let valid = true;

  usernameInput.classList.remove('is-invalid');
  passwordInput.classList.remove('is-invalid');
  roleInput.classList.remove('is-invalid');

  if (!username) {
    usernameInput.classList.add('is-invalid');
    document.getElementById('usernameError').textContent = 'Username is required.';
    valid = false;
  }
  if (password.length < 3) {
    passwordInput.classList.add('is-invalid');
    document.getElementById('passwordError').textContent = 'Password must be at least 3 characters.';
    valid = false;
  }
  if (!role) {
    roleInput.classList.add('is-invalid');
    valid = false;
  }

  if (valid) {
    try {
      const response = await fetch('api/login.php', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ username, password, role })
      });
      const data = await response.json();

      if (data.success) {
        window.location.href = data.redirect_url;
      } else {
        usernameInput.classList.add('is-invalid');
        document.getElementById('usernameError').textContent = data.message || 'Login failed.';
      }
    } catch (err) {
      console.error('Login error:', err);
      alert('Error connecting to authentication API.');
    }
  }
});

            ':full_name' => $data['full_name'],
            ':role'      => $data['role'],
            ':email'     => $data['email'] ?? null
        ]);
        echo json_encode(['status' => 'success', 'message' => 'User created successfully']);
        break;

    case 'toggle_user':
        $data = json_decode(file_get_contents('php://input'), true);
        $stmt = $pdo->prepare("UPDATE users SET is_active = NOT is_active WHERE id = :id");
        $stmt->execute([':id' => $data['user_id']]);
        echo json_encode(['status' => 'success', 'message' => 'Status toggled']);
        break;

    // ----------------------------------------------------
    // 2. INVENTORY MANAGEMENT
    // ----------------------------------------------------
    case 'get_products':
        $stmt = $pdo->query("
            SELECT p.*, s.name AS supplier_name 
            FROM products p 
            LEFT JOIN suppliers s ON p.supplier_id = s.id 
            ORDER BY p.id DESC
        ");
        echo json_encode(['status' => 'success', 'data' => $stmt->fetchAll()]);
        break;

    case 'add_product':
        $data = json_decode(file_get_contents('php://input'), true);
        $stmt = $pdo->prepare("INSERT INTO products (barcode, name, category, price, stock_qty, reorder_level, supplier_id) VALUES (:barcode, :name, :category, :price, :stock, :reorder, :supplier)");
        $stmt->execute([
            ':barcode'  => $data['barcode'],
            ':name'     => $data['name'],
            ':category' => $data['category'] ?? 'General',
            ':price'    => $data['price'],
            ':stock'    => $data['stock_qty'],
            ':reorder'  => $data['reorder_level'] ?? 5,
            ':supplier' => $data['supplier_id'] ?: null
        ]);
        echo json_encode(['status' => 'success', 'message' => 'Product saved successfully']);
        break;

    case 'restock_product':
        $data = json_decode(file_get_contents('php://input'), true);
        $stmt = $pdo->prepare("UPDATE products SET stock_qty = stock_qty + :qty WHERE id = :id");
        $stmt->execute([':qty' => $data['quantity'], ':id' => $data['product_id']]);
        
        $logStmt = $pdo->prepare("INSERT INTO restock_log (product_id, quantity, note) VALUES (:id, :qty, :note)");
        $logStmt->execute([':id' => $data['product_id'], ':qty' => $data['quantity'], ':note' => 'Restocked via admin UI']);
        
        echo json_encode(['status' => 'success', 'message' => 'Restocked successfully']);
        break;

    // ----------------------------------------------------
    // 3. SUPPLIER MANAGEMENT
    // ----------------------------------------------------
    case 'get_suppliers':
        $stmt = $pdo->query("SELECT * FROM suppliers ORDER BY id DESC");
        echo json_encode(['status' => 'success', 'data' => $stmt->fetchAll()]);
        break;

    case 'add_supplier':
        $data = json_decode(file_get_contents('php://input'), true);
        $stmt = $pdo->prepare("INSERT INTO suppliers (name, contact_person, phone, email, address) VALUES (:name, :contact, :phone, :email, :address)");
        $stmt->execute([
            ':name'    => $data['name'],
            ':contact' => $data['contact_person'] ?? null,
            ':phone'   => $data['phone'] ?? null,
            ':email'   => $data['email'] ?? null,
            ':address' => $data['address'] ?? null
        ]);
        echo json_encode(['status' => 'success', 'message' => 'Supplier added']);
        break;

    // ----------------------------------------------------
    // 4. TRANSACTION / SALES / POS
    // ----------------------------------------------------
    case 'checkout':
        $data = json_decode(file_get_contents('php://input'), true);
        $cart = $data['cart'] ?? [];
        $cashierId = $_SESSION['user_id'] ?? null;

        if (empty($cart)) {
            echo json_encode(['status' => 'error', 'message' => 'Cart is empty']);
            exit;
        }

        $pdo->beginTransaction();
        try {
            $subtotal = 0;
            foreach ($cart as $item) {
                $subtotal += $item['price'] * $item['quantity'];
            }
            $taxAmt = $subtotal * 0.08;
            $total  = $subtotal + $taxAmt;

            // 1. Save Transaction
            $txnStmt = $pdo->prepare("INSERT INTO transactions (cashier_id, subtotal, tax_amt, total, payment_method) VALUES (:cashier, :subtotal, :tax, :total, 'cash')");
            $txnStmt->execute([
                ':cashier'  => $cashierId,
                ':subtotal' => $subtotal,
                ':tax'      => $taxAmt,
                ':total'    => $total
            ]);
            $txnId = $pdo->lastInsertId();

            // 2. Save Items & Deduct Stock
            $itemStmt  = $pdo->prepare("INSERT INTO transaction_items (transaction_id, product_id, quantity, unit_price, line_total) VALUES (:tx_id, :prod_id, :qty, :price, :line_total)");
            $stockStmt = $pdo->prepare("UPDATE products SET stock_qty = stock_qty - :qty WHERE id = :prod_id");

            foreach ($cart as $item) {
                $lineTotal = $item['price'] * $item['quantity'];
                $itemStmt->execute([
                    ':tx_id'      => $txnId,
                    ':prod_id'    => $item['product_id'],
                    ':qty'        => $item['quantity'],
                    ':price'      => $item['price'],
                    ':line_total' => $lineTotal
                ]);
                $stockStmt->execute([
                    ':qty'     => $item['quantity'],
                    ':prod_id' => $item['product_id']
                ]);
            }

            $pdo->commit();
            echo json_encode(['status' => 'success', 'message' => 'Checkout complete!', 'transaction_id' => $txnId]);
        } catch (Exception $e) {
            $pdo->rollBack();
            echo json_encode(['status' => 'error', 'message' => 'Transaction failed: ' . $e->getMessage()]);
        }
        break;

    default:
        echo json_encode(['status' => 'error', 'message' => 'Invalid action']);
        break;
}
